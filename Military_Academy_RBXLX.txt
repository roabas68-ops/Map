local function setRank(player, lvl)
	local level = math.clamp(lvl or 1, 1, 10)
	local rData = RANKS[level]
	player:SetAttribute("RankLevel", level)
	player:SetAttribute("RankTitle", rData.Title)

	local team = Teams:FindFirstChild(rData.Team)
	if team then player.Team = team end

	local ls = player:FindFirstChild("leaderstats")
	if ls and ls:FindFirstChild("الرتبة") then
		ls["الرتبة"].Value = rData.Title
	end

	if player.Character then
		createOverhead(player, player.Character, rData)
	end
end
