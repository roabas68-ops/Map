local function findPlayer(prefix)
	local q = string.lower(prefix or "")
	if q == "" then return nil end
	for _, p in ipairs(Players:GetPlayers()) do
		if string.sub(string.lower(p.Name), 1, #q) == q then
			return p
		end
	end
	return nil
end

local function createTag(player, character, rankData)

local function applyRank(player, lvl)
	local level = math.clamp(lvl or 1, 1, 10)
	local data = RANKS[level]
	player:SetAttribute("RankLevel", level)
	player:SetAttribute("RankTitle", data.Title)

	local team = Teams:FindFirstChild(data.Team)
	if team then
		player.Team = team
	end

	local ls = player:FindFirstChild("leaderstats")
	if ls and ls:FindFirstChild("الرتبة") then
		ls["الرتبة"].Value = data.Title
	end

	if player.Character then
		createTag(player, player.Character, data)
	end
end

Players.PlayerAdded:Connect(function(player)
	local ls = Instance.new("Folder")
	ls.Name = "leaderstats"
	ls.Parent = player

	local riyal = Instance.new("IntValue")
	riyal.Name = "ريال"
	riyal.Value = 500
	riyal.Parent = ls

	local rankVal = Instance.new("StringValue")
	rankVal.Name = "الرتبة"
	rankVal.Value = "مدني"
	rankVal.Parent = ls

	local initialRank = 1
	if MilitaryStore then
		local ok, saved = pcall(function()
			return MilitaryStore:GetAsync("Soldier_" .. tostring(player.UserId))
		end)
		if ok and type(saved) == "table" then
			riyal.Value = saved.Riyal or 500
			initialRank = saved.RankLevel or 1
		end
	end

	if OWNERS[player.Name] then
		initialRank = 10
		if riyal.Value < 10000 then
			riyal.Value = 10000
		end
	end

	player.CharacterAdded:Connect(function(char)
		task.wait(0.25)
		applyRank(player, player:GetAttribute("RankLevel") or initialRank)
	end)

	applyRank(player, initialRank)

	task.spawn(function()
		while player.Parent do
			task.wait(60)
			if player.Parent and riyal.Parent then
				local l = player:GetAttribute("RankLevel") or 1
				local pay = (l == 1) and 50 or (100 + l * 25)
				riyal.Value = riyal.Value + pay
			end
		end
	end)

	player.Chatted:Connect(function(msg)
		local senderLvl = player:GetAttribute("RankLevel") or 1
		if senderLvl < 5 and not OWNERS[player.Name] then return end

		local parts = string.split(msg, " ")
		local cmd = string.lower(parts[1] or "")

		if cmd == "!ترقية" and parts[2] then
			local target = findPlayer(parts[2])
			if target then
				local cur = target:GetAttribute("RankLevel") or 1
				if cur < 10 then applyRank(target, cur + 1) end
			end
		elseif cmd == "!تنزيل" and parts[2] then
			local target = findPlayer(parts[2])
			if target then
				local cur = target:GetAttribute("RankLevel") or 1
				if cur > 1 then applyRank(target, cur - 1) end
			end
		elseif cmd == "!اعلان" and #parts >= 2 then
			local txt = table.concat(parts, " ", 2)
			if announceEvent then
				announceEvent:FireAllClients("📢 إعلان عسكري (" .. player.Name .. "): " .. txt)
			end
		elseif cmd == "!سجن" and parts[2] then
			local target = findPlayer(parts[2])
			local world = workspace:FindFirstChild("MilitaryWorld")
			local prison = world and world:FindFirstChild("MilitaryPrison")
			if target and prison and target.Character then
				local hrp = target.Character:FindFirstChild("HumanoidRootPart")
				if hrp then hrp.CFrame = prison.CFrame + Vector3.new(0, 4, 0) end
			end
		end
	end)
end)
