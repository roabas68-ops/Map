local function syncProfile(player, prof)
	local ls = player:FindFirstChild("leaderstats")
	if ls then
		local lvl = ls:FindFirstChild("Level")
		local wins = ls:FindFirstChild("Wins")
		local riyal = ls:FindFirstChild("ريال")
		if lvl then lvl.Value = prof.Level end
		if wins then wins.Value = prof.Wins end
		if riyal then riyal.Value = prof.Riyal end
	end
	UpdateProfileEvent:FireClient(player, prof)
end

local function grantRewards(player, xpGain, riyalGain, wonMatch)
	local prof = Profiles[player.UserId]
	if not prof then return end
	prof.XP = prof.XP + math.max(0, math.floor(xpGain or 0))
	prof.Riyal = prof.Riyal + math.max(0, math.floor(riyalGain or 0))
	if wonMatch then
		prof.Wins = prof.Wins + 1
	end
	local reqXP = prof.Level * 150
	while prof.XP >= reqXP do
		prof.XP = prof.XP - reqXP
		prof.Level = prof.Level + 1
		prof.Riyal = prof.Riyal + 150
		reqXP = prof.Level * 150
	end
	syncProfile(player, prof)
end

local function applyCharacterSpeed(player, charId)
	local cData = findChar(charId)
	local hum = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
	if hum then
		hum.WalkSpeed = math.clamp(16 + ((cData.Stats.Speed - 75) * 0.36), 14, 28)
	end
end

end

SelectCharacterRemote.OnServerInvoke = function(player, charId)
	local prof = Profiles[player.UserId]
	if not prof or type(charId) ~= "string" then
		return false, "طلب غير صالح"
	end
	local cData = findChar(charId)
	local owned = false
	for _, id in ipairs(prof.UnlockedCharacters) do
		if id == charId then
			owned = true
			break
		end
	end
	if not owned then
		local price = (cData.Rarity == "UR") and 1200 or (cData.Rarity == "SSR" and 750 or 450)
		if prof.Riyal < price then
			return false, "رصيد ريال غير كافٍ (" .. tostring(price) .. ")"
		end
		prof.Riyal = prof.Riyal - price
		table.insert(prof.UnlockedCharacters, charId)
	end
	prof.SelectedCharacter = charId
	applyCharacterSpeed(player, charId)
	syncProfile(player, prof)
	return true, "تم تفعيل " .. cData.ArabicName
end

	local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
	if not root or not ball then return end
	if act == "Claim" and (ball.Position - root.Position).Magnitude <= 14 and not ballOwner then
		ballOwner = player
	elseif act == "Pass" and ballOwner == player then
		ballOwner = nil
		player:SetAttribute("BallCD", os.clock() + 0.35)
		ball.AssemblyLinearVelocity = (root.CFrame.LookVector * 80) + Vector3.new(0, 12, 0)
	elseif act == "Shoot" and ballOwner == player then
		ballOwner = nil
		player:SetAttribute("BallCD", os.clock() + 0.45)
		ball.AssemblyLinearVelocity = (root.CFrame.LookVector * 135) + Vector3.new(0, 18, 0)
	elseif act == "Tackle" and (ball.Position - root.Position).Magnitude <= 14 then
		ballOwner = player
		ball.AssemblyLinearVelocity = (root.CFrame.LookVector * 48) + Vector3.new(0, 10, 0)
	end
	broadcastState()
end)
